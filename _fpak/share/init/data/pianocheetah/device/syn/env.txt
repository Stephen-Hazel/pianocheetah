# env.txt - syn's envelopes
# default release envelope MUST be first or you'll have a bad time :)
# to use diff one, it's name must start with rel and be at end of env str.
# put env str at end of track name with leading [ space sepd for multiple envs
#
# env       line starts in 1st col has destination of one of...
#    oscNote oscCent fltCut fltRes amp pan
#
# env stage line starts with space and has
#    level duration curve
#
# level being a real number -1..1
# duration being a string of songtime durations:
#    empty string is dur of 0 for end of envelope
#    w=whole h=half q=quarter e=eighth s=sixteenth 3=32nd 6=64th 1=128th
#    . dotted and 3 triplet
#    eq or e. is a dotted eighth, qqq for 3/4 of a bar, etc
# curve is a real number or "sin"
#    .0001 mostly exponential ..to.. 8 mostly linear
#    default is .001 (exponential/logarithmic)
#    only 1/4 of a sin curve at a time
#    for a full sin, use
#     0 q sin
#     1 q sin
#     0 q sin
#    -1 q sin
#     0 loop
#
# default release - exp curve
release amp
  1 s
  0
#
# do nothin - for multis using |
nop oscCent
  0 w sin
  0 loop
#
# wow (sin) the filter cutoff
wow fltCut
  0 e sin
  1 e sin
  0 loop
#
# phase a 2nd voice (1st voice using nop or whatev)
phaser oscCent
  0 s sin
  3 s sin
  0 loop
#
# pan to the right then stop
panr pan
  0 q
  1
#
# to left
panl pan
  0 q
 -1
#
# pan to right then left and repeat
pano pan
  0 s sin
  1 s sin
  0 s sin
 -1 s sin
  0 loop
#
# to go:  vibr, trem, panran
vibr oscCent
  0   s sin
  20 s sin
  0   s sin
 -20 s sin
  0   loop
